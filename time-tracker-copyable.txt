                    <button class="btn-checkout" id="checkoutBtn" onclick="checkOut()" style="display: none;">Check Out</button>
                </div>

                    <h3>Hours Worked (30 min lunch deducted)</h3>
                    <div id="summaryWorkers"></div>
                </div>

                    <div id="allMaterials"></div>

    <script>
        const workers = ['Adam', 'Worker 1', 'Worker 2'];
        let currentWorker = '';

        function loadData() {
            const saved = localStorage.getItem('timeTrackerData');
            return saved ? JSON.parse(saved) : initializeData();
        }

        function initializeData() {
            const data = {};
            workers.forEach(w => {
                data[w] = {
                    checkinTime: null,
                    checkoutTime: null,
                    location: null,
                    materials: []
                };
            });
            return data;
        }

        let data = loadData();

        function saveData() {
            localStorage.setItem('timeTrackerData', JSON.stringify(data));
        }

        function detectAdmin() {
            const adminToggle = document.getElementById('adminToggle');
            adminToggle.style.display = 'block';
        }

        function switchWorker() {
            const select = document.getElementById('workerSelect');
            currentWorker = select.value;

            if (!currentWorker) {
                document.getElementById('timerCard').style.display = 'none';
                return;
            }

            document.getElementById('timerCard').style.display = 'block';
            updateDisplay();
        }

        function updateDisplay() {
            if (!currentWorker) return;

            const workerData = data[currentWorker];
            const checkinBtn = document.getElementById('checkinBtn');
            const checkoutBtn = document.getElementById('checkoutBtn');
            const statusDisplay = document.getElementById('statusDisplay');
            const statusText = document.getElementById('statusText');
            const notesInput = document.getElementById('notesInput');

            if (workerData.checkinTime && !workerData.checkoutTime) {
                checkinBtn.style.display = 'none';
                checkoutBtn.style.display = 'flex';
                statusDisplay.style.display = 'block';
                statusDisplay.className = 'status checked-in';
                statusText.textContent = `✓ Checked in at ${new Date(workerData.checkinTime).toLocaleTimeString()}`;
                notesInput.disabled = false;
            } else if (workerData.checkoutTime) {
                checkinBtn.style.display = 'flex';
                checkoutBtn.style.display = 'none';
                statusDisplay.style.display = 'block';
                statusDisplay.className = 'status checked-out';
                const hoursWorked = calculateHours(workerData.checkinTime, workerData.checkoutTime);
                statusText.textContent = `✓ Checked out • ${hoursWorked} hours worked`;
                notesInput.disabled = false;
            } else {
                checkinBtn.style.display = 'flex';
                checkoutBtn.style.display = 'none';
                statusDisplay.style.display = 'none';
                notesInput.disabled = true;
            }

            updateMaterialsDisplay();
        }

        function checkIn() {
            if (!currentWorker) return;

            const now = new Date();
            data[currentWorker].checkinTime = now.toISOString();
            data[currentWorker].checkoutTime = null;
            data[currentWorker].location = null;
            saveData();
            updateDisplay();
        }

        function checkOut() {
            if (!currentWorker) return;

            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (position) => {
                        const now = new Date();
                        data[currentWorker].checkoutTime = now.toISOString();
                        data[currentWorker].location = {
                            lat: position.coords.latitude,
                            lng: position.coords.longitude,
                            accuracy: position.coords.accuracy
                        };
                        saveData();
                        updateDisplay();
                        alert('✓ Checked out with location recorded');
                    },
                    (error) => {
                        const now = new Date();
                        data[currentWorker].checkoutTime = now.toISOString();
                        saveData();
                        updateDisplay();
                        alert('✓ Checked out (location unavailable)');
                    }
                );
            } else {
                const now = new Date();
                data[currentWorker].checkoutTime = now.toISOString();
                saveData();
                updateDisplay();
            }
        }

        function calculateHours(checkin, checkout) {
            const start = new Date(checkin);
            const end = new Date(checkout);
            let minutes = (end - start) / 60000;
            
            minutes = Math.max(0, minutes - 30);
            
            const hours = Math.floor(minutes / 60);
            const mins = minutes % 60;
            
            return `${hours}h ${Math.round(mins)}m`;
        }

        function saveMaterials() {
            if (!currentWorker) return;

            const notesText = document.getElementById('notesInput').value;
            if (!notesText.trim()) return;

            const items = notesText
                .split(/[,\n]/)
                .map(item => item.trim())
                .filter(item => item.length > 0);

            data[currentWorker].materials = items;
            saveData();
            updateMaterialsDisplay();
            alert('✓ Materials saved');
        }

        function updateMaterialsDisplay() {
            const materialsDiv = document.getElementById('materialsDisplay');
            const workerData = data[currentWorker];

            if (!workerData.materials || workerData.materials.length === 0) {
                materialsDiv.innerHTML = '';
                return;
            }

            let html = '<div class="materials-list"><h4>Materials to Get:</h4><ul>';
            workerData.materials.forEach(item => {
                html += `<li>${item}</li>`;
            });
            html += '</ul></div>';

            materialsDiv.innerHTML = html;
        }

        function toggleView() {
            const workerView = document.getElementById('workerView');
            const adminView = document.getElementById('adminView');

            if (adminView.classList.contains('active')) {
                adminView.classList.remove('active');
                workerView.style.display = 'block';
            } else {
                adminView.classList.add('active');
                workerView.style.display = 'none';
                updateAdminDashboard();
            }
        }

        function updateAdminDashboard() {
            let summaryHtml = '';
            let allMaterials = {};

            workers.forEach(worker => {
                const wd = data[worker];
                if (wd.checkinTime) {
                    const hoursText = wd.checkoutTime 
                        ? calculateHours(wd.checkinTime, wd.checkoutTime)
                        : 'Still clocked in';

                    let locationHtml = '';
                    if (wd.location) {
                        locationHtml = `<div class="location-badge">📍 ${wd.location.lat.toFixed(4)}, ${wd.location.lng.toFixed(4)}</div>`;
                    }

                    summaryHtml += `
                        <div class="worker-entry">
                            <div class="name">${worker}</div>
                            <div class="hours">${hoursText}</div>
                            ${locationHtml}
                        </div>
                    `;
                }

                if (wd.materials && wd.materials.length > 0) {
                    wd.materials.forEach(item => {
                        if (!allMaterials[item]) {
                            allMaterials[item] = 0;
                        }
                        allMaterials[item]++;
                    });
                }
            });

            document.getElementById('summaryWorkers').innerHTML = summaryHtml || '<p style="color: #999;">No one checked in yet</p>';

            let materialsHtml = '';
            const uniqueMaterials = Object.keys(allMaterials);
            if (uniqueMaterials.length > 0) {
                materialsHtml = '<div class="materials-list" style="border-left-color: #2196F3;"><ul>';
                uniqueMaterials.forEach(item => {
                    materialsHtml += `<li>${item}</li>`;
                });
                materialsHtml += '</ul></div>';
            } else {
                materialsHtml = '<p class="empty-materials">No materials logged yet</p>';
            }

            document.getElementById('allMaterials').innerHTML = materialsHtml;
        }

        function resetDay() {
            if (confirm('Reset all data for today? This cannot be undone.')) {
                data = initializeData();
                saveData();
                location.reload();
            }
        }

        function updateClock() {
            const now = new Date();
            document.getElementById('currentTime').textContent = 
                now.toLocaleTimeString('en-AU', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
        }

        setInterval(updateClock, 1000);
        updateClock();

        detectAdmin();
    </script>
